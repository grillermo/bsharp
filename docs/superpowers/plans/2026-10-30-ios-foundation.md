> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a skeleton "Elisa aprende música" iOS app to Elisa's iPad and the iPhone with one command (`SwiftApp/deploy`). Behind it sits a tested Swift package holding the game's rules, ported from the web app.

**Architecture:** Everything lives under `SwiftApp/`. `SwiftApp/ElisaKit/` is a local Swift package with the pure logic: chords, streak and level-up (including the new piano ↔ guitar switch), adaptive weights, the character deck, plus the OTA quick action and DevLog. It is tested with `swift test` on the Mac. `SwiftApp/App/` is a thin SwiftUI app target generated by XcodeGen from `SwiftApp/project.yml`. `SwiftApp/deploy` (Ruby) builds an ad hoc .ipa and publishes it to files.chiq.me. It is a port of the pipeline already working in `~/c/japanese-walking`.

**Tech Stack:** Swift 6.2 / SwiftUI, iOS 17+, Swift Testing, XcodeGen, Xcode 26.3, Ruby 3.2 (minitest), Python 3 (DevLog server).

**Spec:** `SwiftApp/PLAN.md` (steps 1–3 there). Reference screenshots: `SwiftApp/Reference Material/`.

**The three plans:**
1. **This plan — Foundation:** logic package, skeleton app, OTA deploy, first install on the iPad.
2. **Plan 2 — Playable game:** persistence (`AppState`, session history, JSON import from the web app), audio (`AVAudioEngine`, piano + both guitars), the game screen (flag grid, play/next, answer rules, play hint), note shapes.
3. **Plan 3 — Rewards, menus, switch-over:** character reveal, streak outline, level-up with the instrument switch wired in, the gear menu and its panels, iPad polish, moving Elisa over.

```swift
/// The recordings a chord can be played with. Raw values are the web app's ids, so a profile
/// imported from it keeps its instrument.
public enum Instrument: String, Codable, CaseIterable, Sendable {
    case piano = "piano_1"
    case guitar = "guitar"
    case guitarStrummed = "guitar-strummed"

    /// Folder of this instrument's recordings under `chords/`.
    public var chordFolder: String {
        switch self {
        case .piano: "piano"
        case .guitar: "guitar"
        case .guitarStrummed: "guitar-strummed"
        }
    }

    /// Each level-up switches between piano and guitar, so every new level starts on the
    /// other instrument. Strummed guitar is only ever chosen by hand; it goes back to piano.
    public var afterLevelUp: Instrument {
        self == .piano ? .guitar : .piano
    }
}

/// Correct answers in a row at the current level (src/ts/streak.ts). A miss starts over;
/// reaching the target moves to the next level.
public enum Streak {
    public static let target = 20

    public static func next(_ streak: Int, correct: Bool) -> Int {
        correct ? streak + 1 : 0
    }

    /// How far round the screen outline is filled, 0...1.
    public static func progress(_ streak: Int) -> Double {
        min(1, max(0, Double(streak) / Double(target)))
    }
}

public struct LevelUp: Equatable, Sendable {
    public let level: String
    public let instrument: Instrument
}

public enum Progression {
    /// The level after `current`, or nil when it is the last one (or unknown).
    public static func nextLevel(after current: String, in levels: [String] = Chords.ids) -> String? {
        guard let index = levels.firstIndex(of: current), index < levels.count - 1 else { return nil }
        return levels[index + 1]
    }

    /// Where a completed streak leads: the next level, on the other instrument. Nil at the
    /// last level, where the streak simply starts over and nothing switches.
    public static func levelUp(from level: String, instrument: Instrument) -> LevelUp? {
        nextLevel(after: level).map { LevelUp(level: $0, instrument: instrument.afterLevelUp) }
    }
}
```

```swift
/// Answers per played chord: `matrix[played][chosen]` counts how often `played` sounded and
/// she tapped `chosen`. Counts are Double because older sessions get scaled down.
public typealias ConfusionMatrix = [String: [String: Double]]

/// Session scores (src/ts/stats.ts).
public enum Scoring {
    public static func percentage(correct: Int, identifications: Int) -> Double {
        identifications == 0 ? 0 : 100 * Double(correct) / Double(identifications)
    }

    /// 0...4 for the face shown next to a session: each 10 points above 50% is one step.
    public static func neutralLevel(percentage: Double) -> Int {
        min(max(0, Int(((percentage - 50) / 10).rounded(.down))), 4)
    }
}

/// "Adaptive" chord selection (src/ts/stats.ts): chords she gets wrong, and the chords she
/// mistakes them for, come up more often.
public enum AdaptiveWeights {
    /// Sums `matrices` into one square matrix over `chords`. Chords not in play are dropped.
    public static func merge(_ matrices: [ConfusionMatrix], chords: [String]) -> ConfusionMatrix {
        var out: ConfusionMatrix = Dictionary(uniqueKeysWithValues: chords.map { played in
            (played, Dictionary(uniqueKeysWithValues: chords.map { ($0, 0.0) }))
        })
        for matrix in matrices {
            for (played, row) in matrix {
                for (chosen, count) in row where out[played]?[chosen] != nil {
                    out[played]![chosen]! += count
                }
            }
        }
        return out
    }

    /// One weight per chord in `chords`, summing to 1. A wrong answer adds `wrongWeight` to the
    /// played chord and `mistakenForWeight` to the chosen one. A chord with fewer than
    /// `threshold` answers gets the even share 1/n, and no chord drops below 1/(1.2 n).
    public static func coefficients(
        _ matrix: ConfusionMatrix,
        chords: [String],
        wrongWeight: Double = 5,
        mistakenForWeight: Double = 1.5,
        threshold: Double = 5
    ) -> [Double] {
        let n = chords.count
        let defaultValue = 1 / Double(n)
        let floor = 1 / (1.2 * Double(n))
        var weights = [Double](repeating: 0, count: n)
        var chances = [Double](repeating: 0, count: n)

        for (playedIndex, played) in chords.enumerated() {
            for (chosenIndex, chosen) in chords.enumerated() {
                guard let count = matrix[played]?[chosen] else { continue }
                if chosenIndex != playedIndex {
                    weights[playedIndex] += count * wrongWeight
                    weights[chosenIndex] += count * mistakenForWeight
                } else {
                    weights[playedIndex] += count
                }
                chances[playedIndex] += count
            }
        }

        // Masked weights are fixed; the rest are scaled to fill what is left of 1.
        var mask = chances.map { $0 < threshold }
        weights = weights.indices.map { mask[$0] ? defaultValue : weights[$0] }
        while true {
            weights = normalize(weights, mask: mask)
            guard let low = weights.indices.first(where: { weights[$0] < floor }) else { break }
            mask[low] = true
            weights[low] = floor
        }
        return weights
    }

    private static func normalize(_ values: [Double], mask: [Bool]) -> [Double] {
        let fixed = zip(values, mask).filter { $0.1 }.map(\.0).reduce(0, +)
        let free = zip(values, mask).filter { !$0.1 }.map(\.0).reduce(0, +)
        let factor = free / (1 - fixed)
        return zip(values, mask).map { $1 ? $0 : $0 / factor }
    }

    /// A random item, weighted when `weights` is given (src/ts/utils.ts randomElem).
    /// `items` must not be empty.
    public static func pick<T>(
        _ items: [T],
        weights: [Double]? = nil,
        random: () -> Double = { Double.random(in: 0..<1) }
    ) -> T {
        guard let weights else {
            return items[min(items.count - 1, Int(random() * Double(items.count)))]
        }
        let r = random()
        var cumulative = 0.0
        for (item, weight) in zip(items, weights) {
            cumulative += weight
            if r <= cumulative { return item }
        }
        return items[items.count - 1]
    }
}
```

```swift
/// The Bluey characters shown after a correct answer. Each name is an image in the app's
/// resources (copied from static/characters/ in Plan 3).
public enum Characters {
    public static let all = [
        "dad", "mum", "nana", "grandad", "stripe", "trixie",
        "indy", "chloe", "calypso", "rusty", "lucky", "mackenzie",
        "bluey", "bingo",
        "coco", "jack", "judo", "socks", "honey", "muffin",
        "buddy", "missy", "bently", "winton", "terriers",
    ]

    /// How long a character stays on screen.
    public static let revealSeconds = 3.0

    public static func label(_ name: String) -> String {
        name == "terriers" ? "The Terriers" : name.prefix(1).uppercased() + name.dropFirst()
    }
}

/// Deals every character once per round in random order, and never opens a round with the
/// character that closed the previous one.
public struct CharacterDeck {
    private var deck: [String] = []
    private var last = ""
    private let random: () -> Double

    public init(random: @escaping () -> Double = { Double.random(in: 0..<1) }) {
        self.random = random
    }

    public mutating func next() -> String {
        if deck.isEmpty {
            deck = shuffled(Characters.all)
            if deck.last == last { deck.swapAt(0, deck.count - 1) }
        }
        last = deck.removeLast()
        return last
    }

    // Fisher-Yates driven by `random`, so tests can pin the order.
    private func shuffled(_ items: [String]) -> [String] {
        var out = items
        for i in stride(from: out.count - 1, to: 0, by: -1) {
            out.swapAt(i, Int(random() * Double(i + 1)))
        }
        return out
    }
}
```
